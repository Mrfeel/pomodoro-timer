void algo1(Stack &S){
    int a[10], i, n=0;
    while(!StackEmpty(S)){ n++; Pop(S, a[n]); }
    for(i=1; i<=n; i++) Push(S, a[i]);
}
void algo2(Stack &S, int e){
    Stack T;
    int d;
    InitStack(T);
    while(!EmptyStack(S)){
        Pop(S,d);
        if(d!=e) Push(T,d);
        while(!StackEmpty(T)){
            Pop(T,d);
            Push(S,d);
        }
    }
}
